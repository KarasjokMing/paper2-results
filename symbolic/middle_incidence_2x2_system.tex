\begin{equation}
\begin{pmatrix}
\alpha_{aa} & \alpha_{ab}\\
\alpha_{ba} & \alpha_{bb}
\end{pmatrix}
\begin{pmatrix}
\eta_a^{(3)}\\
\eta_b^{(3)}
\end{pmatrix}
=
\Gamma
\begin{pmatrix}
\mathcal{S}_a^{\rm free}\\
\mathcal{S}_b^{\rm free}
\end{pmatrix}.
\end{equation}

\begin{align}
\alpha_{aa} &= -i\bigl(P_{1,a}^{L}+P_{1,a}^{\rm src}+P_{2,a}^{L}+P_{2,a}^{\rm src}\bigr),\\
\alpha_{ab} &= -i\bigl(P_{2,b}^{L}+e^{i(\varphi_2+\theta_{a2})}+e^{i(\theta_{a2}-\theta_{b2})}\bigr),\\
\alpha_{ba} &= -i\bigl(P_{2,a}^{L}+e^{i(\varphi_2+\theta_{b2})}+e^{i(\theta_{b2}-\theta_{a2})}\bigr),\\
\alpha_{bb} &= -i\bigl(P_{2,b}^{L}+P_{2,b}^{\rm src}+P_{3,b}^{L}+P_{3,b}^{\rm src}\bigr),\\
\mathcal{S}_a^{\rm free} &= 1+e^{i(\varphi_2+\theta_{a2})},\qquad
\mathcal{S}_b^{\rm free} = 1+e^{i(\varphi_2+\theta_{b2})}.
\end{align}
